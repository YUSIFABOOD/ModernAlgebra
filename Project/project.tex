\section{Preliminaries}
In this section, we provide all the theorems and the definitions that we have studied in Modern Algebra, discrete mathematics courses and we used in the following sections. For the sake of time, we omit the proofs.
\defn{Equivalence relation}{equr}
{
    A relation $R$ on a set $A$ is called an \textit{equivalence relation} if it satisfies the following three properties:
    \begin{itemize}
        \item For any $a \in A$, we have that $aRa$ (Reflexivity)
        \item For any $a,x \in A$, we have that if $aRx$, then $xRa$ (Symmetry)
        \item For any $a,x,y \in A$, we have that if $aRx$ and $xRy$, then $aRy$ (Transitivity)
    \end{itemize}
}
\defn{Equivalence class}{equc}
{
    Let $R$ be an equivalence relation on set $A$, and let $a \in A$. The equivalence class of $a$ is the set of all elements in $A$ that are equivalent to $a.$ That is, denoted by $[a]_R$ or $[a].$
    \[[a]=\{b \in A \mid b \sim a\}\]
}
\thm{}{parts}
{
    Suppose $R$ is an equivalence relation on a set $A$. Then we can express $A$ as
    \[A=[x_1]\cup [x_2]\cup [x_3] \cup \cdots \cup [x_n]\]
    where $x_1,x_2,x_3,\cdots,x_n \in A$ and the equivalence classes $A=[x_1]\cup [x_2]\cup [x_3] \cup \cdots \cup [x_n]$ are pairwise disjoint. In other words, $A$ is the union of the distinct equivalence classes of $R$
}
\defn{Center of Group}{Z}{The center of a group $G$ is the set 
\[Z(G)=\{g \in G \mid \forall h \in G, gh=hg\}.\]}
\defn{Centralizer}{cen}
{The Centralizer of an element $g \in G$ is the set $$C(g)=\{a \in G \mid ag=ga\}.$$}
\thm{Lagrange's Theorem}{Lagrange}
{
    The order of a subgroup of a finite group divides the order of the group.
}
\defn{Normal subgroup}{}
{
    A supgroup $N$ of a group $G$ is called \textit{\textbf{normal}} if for any $g \in G$ and any $n \in N$ we have that $gng^{-1}\in N.$ We denote this by $N \trianglelefteq G$ or $N \triangleleft G.$
}
\defn{Normalizer}{}
{
    Let $H$ be a subgroup of a group $G$. The normalizer of $H$, denoted by $N(H)$, is the set of all elements in $G$ that preserve the set $H$ under conjugation.
    $$N(H)=\{g \in G \mid gHg^{-1}=H\}.$$
}
\thm{}{idklol}
{
    Let $Z(G)$ be the center of a group $G.$ if $G/Z(G)$ is cyclic, then $G$ is abelien.
}
\thm{Cauchy's Theorem}{cauchy}
{
    Let $G$ be a finite abelian group, and suppose that $p$ is a prime that divides $|G|$.
Then $G$ has an element of order $p$.
}
\defn{p-Group}{}
{
    Let $p$ be a prime number and let $G$ be a finite group. We call $G$ a \textit{p-Group} if $|G|=p^n$, where $n \ge 1.$
}

\section{Conjugacy Classes}
We start by providing the definition of the Conjugacy class, then show that it is a way to partition groups.
\defn{Conjugacy class}{conj}
{
    Let $a,b$ be elements of group $G$. We say that $a,b$ are \textit{conjugate} in $G$ and we say that $b$ is the \textit{conjugate} of $a$ if and only if $gag^{-1}=b$ for some $g \in G.$ The set of all conjugates of an element $a \in G$, denoted by $\operatorname{cl}(a)=\{gag^{-1} \mid g \in G\}$, is the \textbf{\textit{Conjugacy class of a}}.
}
\thmp{}{conjug}{Conjugacy is an \textit{equivalence relation} on a group.}
{
    Let $R$ be the conjugacy relation and let $G$ be a group.  We show that it is:
    \begin{itemize}
        \item Reflexive: Let $a\in G$. See that $eae^{-1}=a$, so, every element is conjugate to itself and the relation is reflexive.
        \item Symmetric: Let $a,b\in G$. Suppose that $b$ is conjugate of $a,$ that is, there exists $g\in G$ such that $gag^{-1}=b$. Conjugate $b$ by $g^{-1}$, $g^{-1}bg=g^{-1}gag^{-1}g=a.$ Thus, $a$ is conjugate to $b$ and $R$ is symmetric.
        \item Transitive: Let $a,b,c\in G$. Suppose that $a$ is conjugate to $b$, then there exists a $g \in G$ such that $a=gbg^{-1}$, $b$ is conjugate to $c$, then there exists an $h \in G$ such that $b=hch^{-1}.$ Observe that, $a=gbg^{-1}=ghch^{-1}g^{-1}=(gh)c(gh)^{-1},$ and $a$ is conjugate to $c$. Thus, $R$ is transitive.
    \end{itemize}
}
See that, by Theorem \ref{thm:parts}, we can partition groups into disjoint conjugacy classes.
\begin{itemize}
    \item \textbf{Example}\\
In $D_4$, we have:\\
$\operatorname{cl}(R_{90})=\{R_{90}R_{90}R_{90}^{-1},R_{180}R_{90}R_{180}^{-1},R_{270}R_{90}R_{270}^{-1},R_{0}R_{90}R_{0}^{-1},\\ HR_{90}H^{-1},VR_{90}V^{-1},DR_{90}D^{-1},LR_{90}L^{-1}\}=\{R_{90},R_{270} \}$

\end{itemize}
\thmp{}{concen}{$\forall x \in Z(G)$ we have that $\operatorname{cl}(x)=\{x\}.$}
{
    Suppose we have a group $G$, pick an element $a \in Z(G).$ By definition of the center of the group, $a$ commutes with any element in $G.$ Conjugate $a$ by $g\in G,$ that is, $gag^{-1}$. Since $a$ commutes with any element in the group, observe
    \[gag^{-1}=agg^{-1}=ae=a.\]
    Thus, the conjugacy class of an element in the center of the group is the singleton that contains this element.
} 

\mbox{ } \\
The next theorem shows a relationship between the conjugacy class of an element $a$ and the index of the centralizer of $a.$
\thmp{}{number}
{
    Let $G$ be a finite group and let $a$ be an element of $G.$ Then, $|\operatorname{cl}(a)|=|G:C(a)|.$
}
{
    Suppose we have a finite group $G$ and let $a$ be an element of $G$. To prove that the order of the set of conjugates of an element equals the index of that element centralizer in $G$ we need to find a bijection between the set of conjugates and the set of cosets of the centralizer, let $S$ be the set of left cosets of $C(a)$ in $G$.
    Suppose we have the function $\phi: S\to \operatorname{cl}(a)$ such that $\phi(xC(a))=xax^{-1}, x \in G.$ \\ \\
    First, we need to show that $\phi$ is well defined, that is, if $xC(a)=yC(a)$, then $\phi(xC(a))=\phi(yC(a)).$
    Let $x,y$ be elements in the group $G$ and suppose that $xC(a)=yC(a).$ So, $y^{-1}x\in C(a)$ and $x=y(y^{-1}x).$ Hence, since $y^{-1}x$ commutes with $a$ we get that: $$\phi(xC(a))=\phi(y(y^{-1}x)C(a))=y(y^{-1}x)a(y^{-1}x)^{-1}y^{-1}=yay^{-1}.$$
    Since we get that $\phi(xC(a))=\phi(yC(a))$, then the function is well defined.\\
    Next, we show $\phi$ is injective. Suppose $\phi(xC(a))=\phi(yC(a))$, then, $xax^{-1}=yay^{-1}.$ Multiply both sides by $y^{-1}$ from the left and by $x$ from the right. Observe that $y^{-1}xa=ay^{-1}x$, so $y^{-1}x$ commutes with $a$ and $y^{-1}x \in C(a).$ Since $y^{-1}x \in C(a)$ then, $xC(a)=yC(a)$ as required. Thus, the function $\phi$ is injective.
    \\ \\
    Finally, we show that $\phi$ is surjective. Let $b$ be any element in the conjugacy class $\operatorname{cl}(a).$ By definition, $b=gag^{-1}, g \in G.$ We can see the coset $gC(a)$ maps to $b$. Clearly, the function is surjective.
    \\ Therefore, we proved that the function $\phi: S \to \operatorname{cl}(a)$ such that $\phi(xC(a))=xax^{-1}$ is bijective and the order of the conjugacy class of an element $a$ is the index of the centralizer of that element.
}
\corp{}{}
{
    Let $G$ be a finite group and let $a$ be an element in $G$. $|\operatorname{cl}(a)| \text{ divides } |G|, a\in G.$ }
{
    By the previous theorem, we know that $|\operatorname{cl}(a)|=|G:C(a)|$ We know by \hyperref[thm:Lagrange]{Lagrange's Theorem} that the index of a subgroup divides the order of the group and we know that the centralizer of an element in the group is a subgroup. Hence, the order of the conjugacy class of a $\operatorname{cl}(a)$ divides the order of the group $G.$
}
\corp{The Class Equation}{classeq}
{
    Let $G$ be a finite group. $$|G|=\sum |G:C(a)|,$$ where the sum runs over one element $a$ from each conjugacy class of $G$.
}
{
    We know by Theorem \ref{thm:conjug} that the conjugacy classes partition the group, that is, $$|G|=|\operatorname{cl}(x_1)|+ |\operatorname{cl}(x_2)|+\cdots+|\operatorname{cl}(x_n)|,$$ where $x_1,x_2,\cdots, x_n$ are the representatives of the distinct conjugacy classes.
    By Theorem \ref{thm:number}, the index of the centralizer of an element is the order of the conjugacy class of the same element. Thus, we can rewrite the class equation as $|G|=\sum |\operatorname{cl}(a)|.$
    Since the conjugacy class of an element in the center is the singleton that contains the same element, the order of this class is $1$ and the sum of the sizes of these specific classes is exactly the order of the center of the group. Now, we can write the equation
    \[|G|=|Z(G)|+\sum_{a\notin Z(G)}|G:C(a)|.\] 

}
\thmp{}{ntriv}
{
    Let $G$ be a \textit{p-Group}, then $Z(G)$ has more than one element.
}
{
    Suppose the order of a \textit{p-Group} G is $p^n$ for some $n\ge 1.$ We start by collecting the elements in the center, and write the class equation in the form
    \[|G|=|Z(G)|+\sum_{a\notin Z(G)}|G:C(a)|\]
    where the sum runs over all representatives of all conjugacy classes with more than one element.
    We know that $|G:C(a)|=|G|/|C(a)|$, and $C(a)$ is a subgroup of $G$. So each term in $\sum|G:C(a)|$ is of the form $p^k, k\ge 1.$
    Now, observe
    \[|Z(G)|=|G|-\sum|G:C(a)|.\]
    Since $|G|, \sum|G:C(a)|$ are divisible by $p$, then $|Z(G)|$ is divisible by $p$. Since $e \in Z(G)$, $Z(G)$ cannot be empty. Since $|Z(G)|$ is divisble by $p$ and is non-zero, it must be at least $p$. Thus, $Z(G)$ has more than one element.
}
\corp{}{}
{
    if $|G|=p^2$, where $p$ is a prime, then $G$ is \textit{\textbf{abelian}}. 
}   
{
    By Theorem \ref{thm:ntriv} and \hyperref[thm:Lagrange]{Lagrange's Theorem}, $|Z(G)|$ is either $p$ or $p^2.$ If $|Z(G)|=p^2$, $G=Z(G)$ and we are done. Otherwise, suppose $|Z(G)|=p$. Then $|G/Z(G)|=p$ and $G/Z(G)$ is cyclic. By Theorem \ref{thm:idklol}, $G$ is abelian.
}
\section{The Sylow Theorems}
Remember that the inverse of Lagrange's Theorem is not valid. Suppose we have a group $G$ of order $n$, if $m$ divides $n$, this does not necessarily imply that $G$ has a subgroup of order $m.$
However, the Norwegian mathematician Ludwig Sylow (1832-1918) proved the next theorem, which is a partial inverse of Lagrange's Theorem.
\thmp{Sylow's First Theorem (1872)}{sylwfT}
{
    Let $G$ be a finite group and let $p$ be a prime. If $p^k$ divides $|G|$, then $G$ has at least one subgroup of order $p^k.$
}
{
    We proceed by induction,\\
    \textbf{Base case:} Suppose we have a group $G$ and $|G|$ is divisible by $p^0=1.$ It trivially follows that $G$ has one subgroup of order $1$, which is the identity.
    \\
    \textbf{Inductive step:} Assume that the sentence is true for all group of order less than $|G|.$ Then, we have two cases to investigate:
    \begin{itemize}
        \item \textbf{Case 1:} If $G$ has a proper subgroup $H$ such that $p^k \mid |H|$, then, by the inductive hypothesis, $H$ has a subgroup of order $p^k.$ This subgroup of order $p^k$ is also a subgroup of $G$ since $H$ is a subgroup of $G.$
        \item \textbf{Case 2:} Otherwise, assume $G$ does not have a proper subgroup with order divisible by $p^k.$ Observe the class equation:
        \[|G|=|Z(G)|+\sum|G:C(a)|,\] 
        where we run the sum over a representative of each conjugacy class $cl(a)$, and $a \notin Z(G).$
        See that $|G|=|G:C(a)|\cdot|C(a)|$. We know the centralizer is a proper subgroup, so $p^k \nmid |C(a)|$, but $p \mid |G|$, so $p \mid |G:C(a)|.$ From the class equation, $p \mid |Z(G)|.$ By \hyperref[thm:cauchy]{Cauchy's Theorem}, there exist an element $x \in Z(G)$ such that it has order $p.$ Since $x \in Z(G),$ it constructs a cyclic group $\langle x \rangle$ which is normal in $G.$ 
        We know that $|G|$ is divisible by $p^k$, that is, we can write $|G|=mp^k$. Now, we observe that the order of the quotient group $|G/\langle x\rangle|=mp^k/p=mp^{k-1},$ so it is divisible by $p^{k-1}.$ By our inductive hypothesis, $G/\langle x \rangle$ has a subgroup of order $p^{k-1}$ and by Theorem \ref{thm:homot1}, we know that this subgroup has the form $H / \langle x \rangle$, where $H$ is a subgroup of $G.$ Observe that the order of $H/\langle x \rangle = p^{k-1}$ and $|\langle x \rangle|=p$, so $|H|=p^k$. We found a subgroup of order $p^k$, which is a contradiction to our assumption. Therefore, $H$ is the desired subgroup of order $p^k$ and it exists in both cases if $p^k$ divides $|G|$.\end{itemize}
}
By Sylow's First Theorem, if we have a group $G$ such that $|G|= 2^2 \cdot 5^3 \cdot 7^4$, then, $G$ has at least one subgroup of each of the following orders: $2,4,5,25,125,7,49,343,2401.$ Certain subgroups that are guaranteed by Sylow's First Theorem are special in the theory of finite groups. Hence, they are given a special name, \textit{Sylow p-subgroup}.
\defn{Sylow \textit{p-Subgroup}}{}
{
    Let $G$ be a finite group and let $p$ be a prime. if $p^k$ divides $|G|$ and $p^{k+1}$ does not divide $|G|$, then any subgroup of $G$ of order $p^k$ is called a \textit{Sylow p-Subgroup} of $G.$
}
Recall our previous example of $G$ that has order $2^2 \cdot 5^3 \cdot 7^4.$ By \textit{Sylow p-subgroup} definition, we call any subgroup of order $8$ a \textit{Sylow 2-subgroup} of $G$, any subgroup of order $125$ a \textit{Sylow 5-subgroup} of $G$, and so on.
Now, we can generalize Cauchy Theorem to include all finite groups.
\corp{Cauchy's Theorem}{cauchy2}
{
        Let $G$ be a finite group, and suppose that $p$ is a prime divisor of $|G|$.
Then $G$ has an element of order $p$.
}
{
    Suppose we have a finite group $G$ such that $|G|$ is divisible by a prime $p.$ By \hyperref[thm:sylwfT]{Sylow's First Theorem}, we know that $G$ has a subgroup $H$ of order $p.$
    Since $|H|=p$ and $p$ is a prime number, then $H$ is cyclic and generated by a non-identity element. Let $h$ be a generator of $H;$ the order of $h$ is $p$. Thus, we found an element in $G$ of order $p.$ 
}
Before moving to the two more Sylow Theorems, we introduce a new term.
\defn{Conjugate subgroups}{}
{
    Let $H,K$ be subgroup of a group $G.$ We say that $H$ and $K$ are conjugate in $G$ if there is an element $g\in G$ such that $H=gKg^{-1}.$ 
}
\thmp{Sylow's Second Theorem}{sylowsT}
{
    Let $G$ be a finite group and let $H$ be a subgroup of $G.$ If $|H|$ is a power of a prime $p,$ then $H$ is contained in some Sylow p-Subgroup of $G.$
}
{

}
\thmp{Sylow's Third Theorem}{sylowtT}
{
    Let $p$ be a prime and let $G$ be a group of power $p^km$, where $p \nmid m.$ Then the number $n$ of \textit{Sylow p-Subgroups} of $G$ is equal to $1 \mod p$ and divides $m.$ Furthermroe, any two \textit{Sylow p-subgroups} of $G$ are conjugate.
}
